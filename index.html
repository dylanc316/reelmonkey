// ReelMonkey: send-email-updates // Runs every 5 minutes (scheduled from the database). Picks up queued updates // (winning bets, reels passing like milestones) and sends each person one email // covering everything new for them, through Resend. // // Secrets this function needs (Edge Functions > Secrets): // RESEND_API_KEY a Resend API key with sending access // EMAIL_FROM e.g. ReelMonkey (must be on your verified Resend domain) // EMAIL_CRON_SECRET the same random value stored in the database vault as email_cron_secret // SUPABASE_URL and SUPABASE_SERVICE_ROLE_KEY are provided by Supabase automatically. const SITE = "https://reelmonkey.io"; const BATCH = 100; // Resend accepts up to 100 emails per batch request const env = (name) => { const v = Deno.env.get(name); if (!v) throw new Error(`Missing secret ${name}`); return v; }; // ---------- Database (as the service role) ---------- async function rpc(fn, args) { const key = env("SUPABASE_SERVICE_ROLE_KEY"); const res = await fetch(`${env("SUPABASE_URL")}/rest/v1/rpc/${fn}`, { method: "POST", headers: { apikey: key, Authorization: `Bearer ${key}`, "Content-Type": "application/json" }, body: JSON.stringify(args), }); if (!res.ok) throw new Error(`${fn} failed (${res.status}): ${await res.text()}`); return res.json(); } // ---------- Writing the email ---------- const esc = (s) => String(s ?? "").replace(/[&<>"']/g, (c) => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" })[c]); const money = (n) => Number(n).toLocaleString("en-US", { style: "currency", currency: "USD" }); const signedMoney = (n) => (n > 0 ? "+" : n < 0 ? "-" : "") + money(Math.abs(n)); const count = (n) => Number(n).toLocaleString("en-US"); // One update -> a headline, a detail line and where the button goes export function describe(n) { const d = n.data || {}; if (n.kind === "bet_won") { const whose = d.owner_username ? `@${d.owner_username}'s reel` : "a reel"; return { line: `Your ${String(d.side || "").toUpperCase()} bet on ${whose} won ${money(d.payout)}`, detail: `${count(d.target_likes)} likes within ${d.duration_label} · ${signedMoney(Number(d.profit))} profit`, url: `${SITE}/?page=portfolio`, }; } if (n.kind === "likes_milestone") { return { line: `Your reel just passed ${count(d.milestone)} likes`, detail: d.caption ? `“${d.caption}”` : "", url: n.reel_id ? `${SITE}/?reel=${n.reel_id}` : `${SITE}/?page=create`, }; } return null; } // All of one person's updates -> one email export function buildEmail(from, rows) { const items = rows.map(describe).filter(Boolean); if (!items.length) return null; const subject = items.length === 1 ? items[0].line : `You have ${items.length} updates on ReelMonkey`; const button = items.length === 1 ? items[0].url : `${SITE}/?page=portfolio`; const settings = `${SITE}/?page=profile`; const name = rows[0].username; const html = `

ReelMonkey

${name ? `

Hi @${esc(name)},

` : ""} ${items.map((i) => `

${esc(i.line)}

${i.detail ? `

${esc(i.detail)}

` : ""}
`).join("")}

Open ReelMonkey

You're getting this because email updates are on for your ReelMonkey account. Turn them off on your Profile page.

`; const text = [ name ? `Hi @${name},` : "", "", ...items.flatMap((i) => [i.line, i.detail, ""]).filter((l, idx, a) => l !== "" || a[idx - 1] !== ""), `Open ReelMonkey: ${button}`, "", `You're getting this because email updates are on for your ReelMonkey account. Turn them off on your Profile page: ${settings}`, ].join("\n"); return { from, to: [rows[0].email], subject, html, text }; } // ---------- The job ---------- async function run() { const from = env("EMAIL_FROM"); const resendKey = env("RESEND_API_KEY"); const rows = await rpc("claim_email_notifications", { p_limit: 500 }); if (!rows.length) return { claimed: 0, emails: 0, sent: 0 }; // One email per person const byUser = new Map(); for (const r of rows) { if (!byUser.has(r.user_id)) byUser.set(r.user_id, []); byUser.get(r.user_id).push(r); } const emails = []; const unsendable = []; for (const list of byUser.values()) { const email = buildEmail(from, list); if (email) emails.push({ email, ids: list.map((r) => r.id) }); else unsendable.push(...list.map((r) => r.id)); } const sentIds = [...unsendable]; // nothing to say about these, so don't retry them for (let i = 0; i < emails.length; i += BATCH) { const chunk = emails.slice(i, i + BATCH); const res = await fetch("https://api.resend.com/emails/batch", { method: "POST", headers: { Authorization: `Bearer ${resendKey}`, "Content-Type": "application/json" }, body: JSON.stringify(chunk.map((c) => c.email)), }); if (res.ok) chunk.forEach((c) => sentIds.push(...c.ids)); else console.error(`Resend batch failed (${res.status}): ${await res.text()}`); // retried in 15 minutes if (i + BATCH < emails.length) await new Promise((r) => setTimeout(r, 600)); // stay under Resend's rate limit } if (sentIds.length) await rpc("mark_email_notifications_sent", { p_ids: sentIds }); return { claimed: rows.length, emails: emails.length, sent: sentIds.length - unsendable.length }; } Deno.serve(async (req) => { // Only the database's scheduled job knows this value const secret = Deno.env.get("EMAIL_CRON_SECRET"); if (!secret || req.headers.get("x-cron-secret") !== secret) { return new Response("Forbidden", { status: 403 }); } try { return Response.json(await run()); } catch (err) { console.error(err); return Response.json({ error: String(err && err.message || err) }, { status: 500 }); } });
